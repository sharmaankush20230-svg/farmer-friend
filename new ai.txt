  async function startCamera() {
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {

    const file = event.target.files && event.target.files[0];
    event.target.value = "";
    if (!file) return;

    if (!file.type || file.type.indexOf("image/") !== 0) {

      return;
    }
    if (file.size > 15 * 1024 * 1024) {
      showAlert("Image is too large (over 15 MB). Please choose a smaller photo.", "error");
      return;
    }
    hideAlert();
    processImageFile(file);
  }

  function processImageFile(file) {
    const reader = new FileReader();
    reader.onerror = function () {
      showAlert("Could not read that file. Please try a different image.", "error");
    };
    reader.onload = function (e) {
      const img = new Image();
      img.onerror = function () {
        showAlert("That image appears to be corrupted or unsupported.", "error");
      };
      img.onload = function () {
        const canvas = document.createElement("canvas");
        let { width, height } = img;

        if (width > height && width > MAX_DIMENSION) {
          height = Math.round((height * MAX_DIMENSION) / width);
          width = MAX_DIMENSION;
        } else if (height > MAX_DIMENSION) {
          width = Math.round((width * MAX_DIMENSION) / height);
          height = MAX_DIMENSION;
        }

        canvas.width = width;
        canvas.height = height;
        const ctx = canvas.getContext("2d");
        ctx.fillStyle = "#ffffff";
        ctx.fillRect(0, 0, width, height);
        ctx.drawImage(img, 0, 0, width, height);

        const dataUrl = canvas.toDataURL("image/jpeg", 0.85);

        currentMimeType = "image/jpeg";
        currentImageBase64 = dataUrl.split(",")[1] || null;

        previewImg.src = dataUrl;
        previewWrap.classList.add("visible");
        previewMeta.textContent = "Image ready · " + width + "×" + height + " px";
        resultsCard.classList.remove("visible");
        hideAlert();
      };
      img.src = e.target.result;
    };
    reader.readAsDataURL(file);
  }

  clearBtn.addEventListener("click", function () {
    currentImageBase64 = null;
    previewImg.removeAttribute("src");
    previewWrap.classList.remove("visible");
    previewMeta.textContent = "Ready to analyze";
    resultsCard.classList.remove("visible");
    hideAlert();
    setStatus("");
  });

  /* ---------- JSON parsing + rendering ---------- */
  function extractJson(text) {
    if (!text) return null;
    let cleaned = String(text).trim();
    const fenceMatch = cleaned.match(/```(?:json)?\s*([\s\S]*?)```/i);
    if (fenceMatch) cleaned = fenceMatch[1].trim();
    const first = cleaned.indexOf("{");
    const last = cleaned.lastIndexOf("}");
    if (first !== -1 && last !== -1 && last > first) {
      cleaned = cleaned.slice(first, last + 1);
    }
    try { return JSON.parse(cleaned); } catch (e) { return null; }
  }

  function setResultText(el, value, fallback) {
    const text = (value === null || value === undefined || String(value).trim() === "")
      ? fallback
      : String(value).trim();
    el.textContent = text;
    if (text === fallback) el.classList.add("empty");
    else el.classList.remove("empty");
  }

  function renderResults(data) {
    const plantName = data.plant_name || data.plantName || "Unknown plant";
    const status = (data.health_status || data.healthStatus || "Unknown").trim();
    const disease = data.disease_name || data.diseaseName || "";
    const confidence = data.confidence_level || data.confidenceLevel || "";
    const symptoms = data.symptoms || "";
    const organic = data.organic_cure || data.organicCure || data.organic_treatment || "";
    const chemical = data.chemical_cure || data.chemicalCure || data.chemical_treatment || "";

    resPlantName.textContent = plantName;

    const lower = status.toLowerCase();
    resStatusBadge.classList.remove("healthy", "diseased", "unknown");
    if (lower.indexOf("healthy") !== -1 && lower.indexOf("diseased") === -1) {
      resStatusBadge.classList.add("healthy");
      resStatusBadge.textContent = "Healthy";
    } else if (lower.indexOf("diseas") !== -1 || lower.indexOf("infect") !== -1 || lower.indexOf("pest") !== -1) {
      resStatusBadge.classList.add("diseased");
      resStatusBadge.textContent = status || "Diseased";
    } else {
      resStatusBadge.classList.add("unknown");
      resStatusBadge.textContent = status || "Unknown";
    }

    setResultText(resDisease, disease, "None detected");
    setResultText(resSymptoms, symptoms, "No symptoms described.");
    setResultText(resOrganic, organic, "No organic treatment suggested.");
    setResultText(resChemical, chemical, "No chemical treatment suggested.");

    if (confidence) {
      resConfidence.style.display = "inline-block";
      resConfidence.textContent = "Confidence: " + confidence;
    } else {
      resConfidence.style.display = "none";
      resConfidence.textContent = "";
    }

    resultsCard.classList.add("visible");
    try { resultsCard.scrollIntoView({ behavior: "smooth", block: "start" }); } catch (e) {}
  }

  /* ---------- Analyze ---------- */
  async function analyze() {
    if (isAnalyzing) return;
    hideAlert();

    const apiKey = getApiKey();
    if (!apiKey) {
      showAlert("API key is missing. Please open the file and paste your Gemini API key on the GEMINI_API_KEY line.", "error");
      return;
    }
    if (!currentImageBase64) {
      showAlert("Please capture or upload a photo of a leaf first.", "warn");
      return;
    }

    isAnalyzing = true;
    captureBtn.disabled = true;
    resultsCard.classList.remove("visible");

    const steps = ["Scanning leaves…", "Analyzing pathology…", "Consulting Gemini AI…", "Preparing recommendations…"];
    let stepIndex = 0;
    setStatus(steps[0]);
    const stepTimer = setInterval(function () {
      stepIndex = (stepIndex + 1) % steps.length;
      setStatus(steps[stepIndex]);
    }, 1400);

    const prompt =
      "You are an expert agricultural plant pathologist. Analyze the leaf image provided. " +
      "Identify the plant species, determine whether it is healthy or diseased, name the specific " +
      "disease or pest if present, describe the visible symptoms, and provide actionable organic " +
      "and chemical treatment or care recommendations. " +
      "Respond with ONLY a single valid JSON object (no markdown, no code fences, no extra text) " +
      "using exactly these keys: " +
      "plant_name, health_status, disease_name, confidence_level, symptoms, organic_cure, chemical_cure. " +
      "Use short, clear, farmer-friendly sentences.";

    const payload = {
      contents: [
        {
          parts: [
            { text: prompt },
            { inlineData: { mimeType: currentMimeType, data: currentImageBase64 } }
          ]
        }
      ]
    };

    try {
      const response = await fetch(API_URL + encodeURIComponent(apiKey), {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify(payload)
      });

      if (!response.ok) {
        let detail = "";
        try {
          const errJson = await response.json();
          if (errJson && errJson.error && errJson.error.message) detail = errJson.error.message;
          else detail = JSON.stringify(errJson);
        } catch (e) {
          try { detail = await response.text(); } catch (e2) { detail = ""; }
        }

        let userMsg = "Gemini API error (HTTP " + response.status + ").";
        if (response.status === 400) {
          userMsg = "Bad request — check your API key or model name. " + (detail ? "Details: " + detail : "");
        } else if (response.status === 401 || response.status === 403) {
          userMsg = "API key rejected (HTTP " + response.status + "). Double-check the key. " + (detail ? "Details: " + detail : "");
        } else if (response.status === 404) {
          userMsg = "Model '" + MODEL_ID + "' not found. Try 'gemini-2.5-flash' in the MODEL_ID line. " + (detail ? "Details: " + detail : "");
        } else if (response.status === 429) {
          userMsg = "Rate limit hit (429). Please wait a minute and retry. " + (detail ? "Details: " + detail : "");
        } else if (response.status === 503) {
          userMsg = "Gemini servers are busy (503). Try again in a few moments.";
        } else if (response.status >= 500) {
          userMsg = "Gemini service temporarily unavailable (HTTP " + response.status + "). Please retry.";
        }
        throw new Error(userMsg);
      }

      const json = await response.json();

      let modelText = "";
      if (
        json && Array.isArray(json.candidates) && json.candidates[0] &&
        json.candidates[0].content && Array.isArray(json.candidates[0].content.parts)
      ) {
        for (const part of json.candidates[0].content.parts) {
          if (typeof part.text === "string") modelText += part.text;
        }
      }

      if (!modelText.trim()) {
        const blockReason =
          (json && json.promptFeedback && json.promptFeedback.blockReason) ||
          (json && json.candidates && json.candidates[0] && json.candidates[0].finishReason) ||
          "";
        throw new Error(
          "Gemini returned an empty response" +
          (blockReason ? " (reason: " + blockReason + ")." : ".") +
          " Try a clearer photo of a single leaf."
        );
      }

      const parsed = extractJson(modelText);
      if (!parsed) {
        throw new Error("Could not parse AI response as JSON. Try again.");
      }

      renderResults(parsed);
    } catch (err) {
      console.error("Analysis failed:", err);
      const msg = (err && err.message) ? err.message : "Something went wrong while contacting Gemini.";
      showAlert(msg, "error");
    } finally {
      clearInterval(stepTimer);
      setStatus("");
      isAnalyzing = false;
      captureBtn.disabled = !videoStream;
    }
  }
})();
